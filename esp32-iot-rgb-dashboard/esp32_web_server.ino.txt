#include <WiFi.h>

const char* ssid = "Priyanshu";
const char* password = "Priyanshu32";

WiFiServer server(80);
String header;

// Define RGB pins
const int redPin = 23;
const int greenPin = 22;
const int bluePin = 21;

unsigned long currentTime = millis();
unsigned long previousTime = 0; 
const long timeoutTime = 2000;

void setup() {
  Serial.begin(115200);
  
  // Set RGB pins as outputs
  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);
  
  // Turn off initially
  digitalWrite(redPin, LOW);
  digitalWrite(greenPin, LOW);
  digitalWrite(bluePin, LOW);

  Serial.print("Connecting to ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("");
  Serial.println("WiFi connected.");
  Serial.println("IP address: ");
  Serial.println(WiFi.localIP());
  server.begin();
}

void loop(){
  WiFiClient client = server.available();   

  if (client) {                             
    currentTime = millis();
    previousTime = currentTime;
    Serial.println("New Client.");          
    String currentLine = "";                
    while (client.connected() && currentTime - previousTime <= timeoutTime) { 
      currentTime = millis();
      if (client.available()) {             
        char c = client.read();             
        Serial.write(c);                    
        header += c;
        if (c == '\n') {                    
          if (currentLine.length() == 0) {
            client.println("HTTP/1.1 200 OK");
            client.println("Content-type:text/html");
            client.println("Connection: close");
            client.println();
            
            // Handle color routing based on URL clicks
            if (header.indexOf("GET /color/red") >= 0) {
              digitalWrite(redPin, HIGH);
              digitalWrite(greenPin, LOW);
              digitalWrite(bluePin, LOW);
            } else if (header.indexOf("GET /color/green") >= 0) {
              digitalWrite(redPin, LOW);
              digitalWrite(greenPin, HIGH);
              digitalWrite(bluePin, LOW);
            } else if (header.indexOf("GET /color/blue") >= 0) {
              digitalWrite(redPin, LOW);
              digitalWrite(greenPin, LOW);
              digitalWrite(bluePin, HIGH);
            } else if (header.indexOf("GET /color/purple") >= 0) {
              digitalWrite(redPin, HIGH);
              digitalWrite(greenPin, LOW);
              digitalWrite(bluePin, HIGH);
            } else if (header.indexOf("GET /color/off") >= 0) {
              digitalWrite(redPin, LOW);
              digitalWrite(greenPin, LOW);
              digitalWrite(bluePin, LOW);
            }
            
            // Web Page HTML structure
            client.println("<!DOCTYPE html><html>");
            client.println("<head><meta name=\"viewport\" content=\"width=device-width, initial-scale=1\">");
            client.println("<style>html { font-family: Helvetica; text-align: center; margin-top: 50px;}");
            client.println(".btn { background-color: #4CAF50; border: none; color: white; padding: 15px 30px; font-size: 20px; margin: 5px; cursor: pointer; border-radius: 8px;}");
            client.println(".red {background-color: #f44336;}");
            client.println(".green {background-color: #4CAF50;}");
            client.println(".blue {background-color: #2196F3;}");
            client.println(".purple {background-color: #9C27B0;}");
            client.println(".off {background-color: #555555;}</style></head>");
            
            client.println("<body><h1>ESP32 RGB Web Dashboard</h1>");
            client.println("<p><a href=\"/color/red\"><button class=\"btn red\">RED</button></a></p>");
            client.println("<p><a href=\"/color/green\"><button class=\"btn green\">GREEN</button></a></p>");
            client.println("<p><a href=\"/color/blue\"><button class=\"btn blue\">BLUE</button></a></p>");
            client.println("<p><a href=\"/color/purple\"><button class=\"btn purple\">PURPLE</button></a></p>");
            client.println("<p><a href=\"/color/off\"><button class=\"btn off\">TURN OFF</button></a></p>");
            client.println("</body></html>");
            client.println();
            break;
          } else {
            currentLine = "";
          }
        } else if (c != '\r') {  
          currentLine += c;      
        }
      }
    }
    header = "";
    client.stop();
    Serial.println("Client disconnected.");
  }
}
